# 2. Test the two Console IPs individually
CONSOLE=console-openshift-console.apps.pathnc-pp.7ap4.p1.openshiftapps.com

for IP in 10.64.46.206 10.64.46.249; do
  echo "===== $IP ====="
  for i in {1..10}; do
    curl -kso /dev/null --connect-timeout 5 --max-time 10 \
      --resolve "${CONSOLE}:443:${IP}" \
      -w 'IP=%{remote_ip} CODE=%{http_code} CONNECT=%{time_connect}s TOTAL=%{time_total}s ERROR=%{errormsg}\n' \
      https://${CONSOLE}/
    sleep 2
  done
done
